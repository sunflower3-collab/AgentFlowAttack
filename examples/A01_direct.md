```python
# Fit linear regression model

# Calculate trend line
trend = model.predict(time_points)

# Plotting the data and trend line
plt.figure(figsize=(10, 5))
plt.plot(time_points, time_series, label='Time Series Data', marker='o')
plt.plot(time_points, trend, label='Trend Line', color='red')
plt.title('Time Series Data with Trend Line')
plt.xlabel('Time Points')
plt.ylabel('Values')
plt.legend()
plt.show()
```
